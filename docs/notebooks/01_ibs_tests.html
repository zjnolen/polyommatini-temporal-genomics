Investigating reference bias with different aligners

Plot comparisons of reference bias for historical and modern samples when aligned to the linear reference with bwa mem/aln and when aligned to a variation graph augmented with called modern variants with vg.

ibs_all <- c()
het_all <- c()

for (i in c(1:length(species))) {
  s <- species[i]
  g <- genomes[i]
  for (m in c("vg-notrans", "bwa-notrans")) {
    ibs <- fread(
      paste0(
        "../analyses/results/datasets/historical-",
        s,
        "-",
        m,
        "/qc/ibs_refbias/historical-",
        s,
        "-",
        m,
        ".",
        g,
        "_all_allsites-filts.refibs.tsv"
      ),
      header = TRUE,
      sep = "\t",
      select = c("sample", "ibs.to.ref")
    )
    ibs$type <- "random"
    ibs2 <- read.table(
      paste0(
        "../analyses/results/datasets/historical-",
        s,
        "-",
        m,
        "/bcfs/historical-",
        s,
        "-",
        m,
        ".",
        g,
        "_all.dp6_allsites-filts.filtered_mindp5-allsites-jointcall_allbal0.21-0.79.notrans.fmiss0.3.bcf.stats.ref_bias"
      ),
      header = FALSE)
    colnames(ibs2) <- c("sample", "ibs.to.ref")
    ibs2$type <- "genotypes"
    ibs <- rbind(ibs, ibs2)
    het <- read.table(
      paste0(
        "../analyses/results/datasets/historical-",
        s,
        "-",
        m,
        "/analyses/heterozygosity/historical-",
        s,
        "-",
        m,
        ".",
        g,
        "_all.dp6_allsites-filts.filtered_mindp5-allsites-jointcall_allbal0.21-0.79.notrans.fmiss0.3.bcftools.het.tsv"
      ),
      header = TRUE
    )
    het2 <- read.table(
      paste0(
        "../analyses/results/datasets/historical-",
        s,
        "-",
        m,
        "/analyses/heterozygosity/historical-",
        s,
        "-",
        m,
        ".",
        g,
        "_all_allsites-filts_heterozygosity.tsv"
      ),
      header = TRUE
    )
    het <- het[ , c(1:2)]
    het[ , 2] <- het[ , 2]*1000
    colnames(het) <- c("sample", "het")
    het2 <- het2[ , c(1, 3)]
    colnames(het2) <- c("sample", "het")
    het$type <- "Genotype Calls"
    het2$type <- "Genotype Likelihoods"
    het <- rbind(het, het2)
    qc <- fread(
      paste0(
        "../analyses/results/datasets/historical-",
        s,
        "-",
        m,
        "/qc/historical-",
        s,
        "-",
        m,
        ".",
        g,
        "_all.sampleqc.tsv"
      ),
      header = TRUE,
      sep = "\t",
      select = c(
        "sample", "population", "time"
      )
    )
    ibs <- merge(ibs, qc, by = c("sample"))
    ibs$group <- m
    ibs$species <- s
    het <- merge(het, qc, by = c("sample"))
    het$group <- m
    het$species <- s
    ibs_all <- rbind(ibs_all, ibs)
    het_all <- rbind(het_all, het)
  }
}


ibs_all$time <- factor(ibs_all$time,
  levels = c("historical", "modern"),
  labels = c("Historical", "Modern")
)

ibs_all$species <- factor(ibs_all$species,
  levels = c("picarus", "pargus", "csemiargus"),
  labels = c("Po. icarus", "Pl. argus", "Cy. semiargus")
)

ibs_all$aligner <- ifelse(grepl("vg", ibs_all$group), "vg", "bwa")

ibs_all$aligner <- factor(ibs_all$aligner,
  levels = c("bwa", "vg"),
  labels = c("BWA", "VG"))


randoms <- ggplot(ibs_all[ibs_all$type == "random", ], aes(x = time, y = ibs.to.ref)) +
