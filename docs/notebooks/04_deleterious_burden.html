Putatively deleterious mutation burden

Derived count statistics and table

In this study as a genomic proxy for genetic load, we used the derived count of putatively deleterious mutations. The main workflow provides this output as a table of derived allele counts at different position categories, including counts in homozygous state in particular. We first compile these counts into derived counts / 1000 genotyped positions to make them comparable across samples with variable missingness. We calculate means of these counts for each mutation category both for the total count and homozygous count. Then we performa linear mixed models with derived count as a response to sampling time period, with sampling region as a random factor. We also get confidence intervals around the mean counts with jackknifing.

# read in the derived count tables as output by the workflow, using the one
# that is polarized by the inferred ancestral states. read in Froh estimates
# as well to associate them later
varcounts_all <- c()
froh_all <- c()

for (i in c(1:length(species))) {
  s <- species[i]
  g <- genomes[i]

  # Read in variant impact counts
  varcounts <- read.table(
    paste0(
      "../analyses/results/datasets/historical-",
      s,
      "-vg-notrans/analyses/burden/historical-",
      s,
      "-vg-notrans.",
      g,
      "_all.dp6_allsites-filts.filtered_mindp5-biallelic-jointcall_allbal0.21-0.79.notrans.fmiss0.3.varimpacts_anc.tsv"
    ),
    header = TRUE
  )

  froh <- fread(
    paste0(
      "../analyses/results/datasets/historical-",
      s,
      "-vg-notrans/plots/inbreeding/historical-",
      s,
      "-vg-notrans.",
      g,
      "_all.dp6_allsites-filts.filtered_mindp5-biallelic-jointcall_allbal0.21-0.79.notrans.fmiss0.3.bcftools.ind_froh.tsv"
    ),
    header = TRUE,
    sep = "\t",
    select = c("sample", "population", "time", "depth", "Froh")
  )
  varcounts <- merge(varcounts, froh, by = c("sample", "population", "time", "depth"))
  varcounts$species <- s
  froh$species <- s
  froh_all <- rbind(froh_all, froh)
  varcounts_all <- rbind(varcounts_all, varcounts)
}

varcounts_all$year <- str_sub(varcounts_all$population, -4, -1)
varcounts_all$region <- str_sub(varcounts_all$population, 1, -5)

varcounts_all$species <- factor(

varcounts_all$region_group <- varcounts_all$region
varcounts_all[varcounts_all$region_group == "SW Skåne", ]$region_group <- "W Skåne"
varcounts_all[varcounts_all$region_group == "SE Skåne", ]$region_group <- "E Skåne"

# perform linear models for derived count as a response to time period across
# relevant categories of mutations. these include putatively deleterious
# categories (high impact, moderate impact, conserved) as well as putatively
# neutral categories (low impact, intergenic) as a control. always normalize
# by total genotype count (nsites)
varcounts_stats <- c()

for (s in species_names) {
  for (i in c("gerp_n", "high", "mod", "low", "inter", "gerp_hom_n", "high_hom", "mod_hom", "low_hom", "inter_hom")) {
    hist <- mean(varcounts_all[varcounts_all$species == s & varcounts_all$time == "Historical", i] * 1000 / varcounts_all[varcounts_all$species == s & varcounts_all$time == "Historical", "nsites"])
    mod <- mean(varcounts_all[varcounts_all$species == s & varcounts_all$time == "Modern", i] * 1000 / varcounts_all[varcounts_all$species == s & varcounts_all$time == "Modern", "nsites"])
    ratio <- mod / hist
    model <- lme(as.formula(paste0(i, " * 1000 / nsites ~ time")),
      random = ~ 1 | region_group,
      data = varcounts_all[varcounts_all$species == s, ]
    )
    tidyed <- tidy(model, conf.int = TRUE, effects = "fixed")
    tidydf <- tidyed[2, c(3, 4, 6, 7, 8, 9)]
    tidydf$hist.mean <- hist
    jackknife <- quantile(jackknife(varcounts_all[varcounts_all$species == s & varcounts_all$time == "Historical", i] * 1000 / varcounts_all[varcounts_all$species == s & varcounts_all$time == "Historical", "nsites"], mean)$jack.values, c(0.025, 0.975))
    tidydf$hist.low.jk.ci <- jackknife[1]
    tidydf$hist.high.jk.ci <- jackknife[2]
    tidydf$mod.mean <- mod
    jackknife <- quantile(jackknife(varcounts_all[varcounts_all$species == s & varcounts_all$time == "Modern", i] * 1000 / varcounts_all[varcounts_all$species == s & varcounts_all$time == "Modern", "nsites"], mean)$jack.values, c(0.025, 0.975))
    tidydf$mod.low.jk.ci <- jackknife[1]
    tidydf$mod.high.jk.ci <- jackknife[2]
    tidydf$`mod/hist` <- ratio
    tidydf <- round(tidydf, 5)
    tidydf$species <- s
    tidydf <- tidydf[, c(14, 7, 8, 9, 10, 11, 12, 13, 1, 2, 3, 5, 6, 4)]
    tidydf$type <- i
    varcounts_stats <- rbind(varcounts_stats, tidydf)
  }
}

Inbreeding coefficients partitiond by RoH size

To visualize how much of the inbreeding coefficient was composed by runs of homozygosity by different lengths, we plotted them as bars colored by the proportion coming from different size classes.

rohs <- read.table("../analyses/results/datasets/historical-csemiargus-vg-notrans/analyses/roh/bcftools/historical-csemiargus-vg-notrans.ilCyaSemi1.1_all.dp6_allsites-filts.filtered_mindp5-biallelic-jointcall_allbal0.21-0.79.notrans.fmiss0.3.regs.roh")

colnames(rohs) <- c(
  "type", "sample", "chr", "start", "end", "length",
  "inform_sites", "phred"
)

# exclude ROH < 200kb and < 30 Pherd score
rohs <- rohs[rohs$length >= 200000 & rohs$phred >= 30, ]

nroh <- rohs %>%
  group_by(sample) %>%
  summarise(cumroh = sum(length), nroh = n())

inds <- read.table(
  "../analyses/results/datasets/historical-csemiargus-vg-notrans/poplists/historical-csemiargus-vg-notrans_all.indiv.list",
  header = TRUE
)

# drop uncalled sample from ind list for sample looping
inds <- inds[inds$sample != "MZLU152765", ]

nroh <- merge(inds, nroh, by = "sample", all = TRUE)
nroh[is.na(nroh)] <- 0

norun <- c()
norun$type <- rep("RG", nrow(inds))
norun$sample <- inds$sample
norun$chr <- rep(0, nrow(inds))
norun$start <- rep(0, nrow(inds))
norun$end <- rep(0, nrow(inds))
norun$length <- rep(0, nrow(inds))
norun$inform_sites <- rep(0, nrow(inds))
norun$phred <- rep(0, nrow(inds))
norun <- data.frame(norun)

# set total autosomal length for C. semiargus
autosize <- 416794061

# prepare size bins to loop over
bins <- c(Inf, 6000000, 3000000, 1500000, 500000, 200000)

frohs <- c()
ranges <- c()

# loop over bins, quantifying Froh for that size bin in each sample
for (i in seq(length(bins))) {
  if (i <= length(bins) - 1) {
    runs <- rohs[rohs$length < bins[i], ]
    runs <- runs[runs$length >= bins[i + 1], ]
    runs <- rbind(runs, norun)
    runs <- runs %>%
      group_by(sample) %>%
      summarize(length = sum(length))
    runs <- merge(runs, inds, by = "sample")
    runs$froh <- runs$length / autosize
    runs$range <- paste0(bins[i], "-", bins[i + 1])
    ranges <- c(ranges, paste0(bins[i], "-", bins[i + 1]))
    frohs <- rbind(frohs, runs)
  }
}
