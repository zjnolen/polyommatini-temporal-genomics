Heterozygosity as a response to time period statistics

Run statistical tests on heterozygosity - performing LMMs with heterozygosity as a response to time period with population as a random factor. Also perform jackknifes and bootstraps to get confidence intervals around the mean estimates.

# get means and confidence intervals around mean heterozygosity using
# bootstrapping of samples

hz_means <- hz_all %>%
  group_by(species, population) %>%
  do(
    data.frame(
      rbind(
        Hmisc::smean.cl.boot(.$hetperbp, B = 1000)
      )
    )
  )

write.table(hz_means, "../tables/hz_means.tsv",
  row.names = FALSE, quote = FALSE, sep = "\t"
)

# prepare table for supplement with historical/modern heterozygosity comparisons

hz_stats <- c()

for (s in species_names) {
  hist <- mean(hz_all[hz_all$species == s & hz_all$time == "Historical", ]$hetperbp * 1000)
  mod <- mean(hz_all[hz_all$species == s & hz_all$time == "Modern", ]$hetperbp * 1000)
  # estimate ratio of modern to historical heterozygosity
  ratio <- mod/hist
  # run linear mixed model with heterozygosity as response to time period and pop as a random eff.
  model <- lme(hetperbp * 1000 ~ time, random = ~1|region_group, data = hz_all[hz_all$species == s, ])
  # extract model parameter estimates with tidy
  tidyed <- tidy(model, conf.int = TRUE, effects = "fixed")
  tidydf <- tidyed[2, c(3,4,6,7,8,9)]
  tidydf$hist.mean <- hist
  # perform jackknifing as another approach to confidence intervals around means
  jackknife <- quantile(jackknife(hz_all[hz_all$species == s & hz_all$time == "Historical", ]$hetperbp * 1000, mean)$jack.values, c(0.025, 0.975))
  tidydf$hist.low.jk.ci <- jackknife[1]
  tidydf$hist.high.jk.ci <- jackknife[2]
  tidydf$mod.mean <- mod
  jackknife <- quantile(jackknife(hz_all[hz_all$species == s & hz_all$time == "Modern", ]$hetperbp * 1000, mean)$jack.values, c(0.025, 0.975))
  tidydf$mod.low.jk.ci <- jackknife[1]
  tidydf$mod.high.jk.ci <- jackknife[2]
  tidydf$`mod/hist` <- ratio
  tidydf <- round(tidydf, 5)
  tidydf$species <- s
  tidydf <- tidydf[ , c("species", "hist.mean", "hist.low.jk.ci", "hist.high.jk.ci", "mod.mean", "mod.low.jk.ci", "mod.high.jk.ci", "mod/hist", "estimate", "std.error", "statistic", "conf.low", "conf.high", "p.value")]
  hz_stats <- rbind(hz_stats, tidydf)
}

write.table(hz_stats, "../tables/hz_stats.tsv",
  row.names = FALSE, quote = FALSE, sep = "\t"
)

Coalescent simulations

These were done with Slendr using the msprime backend, manually running them with the Rscript found at analyses/slendr/butterfly-decline-predict.R. This code summarizes this data into distributions of simulated outcomes reported in Table S10 and Figure S9.
