Heterozygosity change per year

One of the metrics suggested for utilization in Sweden is the per year change in heterozygosity. While the warning categories are suggested only for significant differences, which we can’t really check with single samples per time period, we use this same calculation as it is a useful visualization.

# Calculate per species heterozygosity retention pairwise between all historical
# and modern individual pairs in each species and report a median and range

relhet_all <- c()

  for (i in seq_along(c(main_species, species))) {
    s <- c(main_species, species)[i]
    f <- c(main_species_f, species_f)[i]
    g <- c(main_genomes, genomes)[i]
    m <- c(main_species_m, species_m)[i]

    qc <- fread(
      paste0(
        "../analyses/results/datasets/historical-",
        f,
        "/qc/historical-",
        f,
        ".",
        g,
        "_all.sampleqc.tsv"
      ),
      header = TRUE,
      sep = "\t"
    )
    qc <- qc[, c("sample", "allsites-filts.depth.mean")]
    hz <- fread(
      paste0(
        "../analyses/results/datasets/historical-",
        f,
        "/analyses/heterozygosity/historical-",
        f,
        ".",
        g,
        "_all.dp6_allsites-filts.filtered_mindp5-allsites-jointcall_allbal0.21-0.79.notrans.fmiss",
        m,
        ".bcftools.het.tsv"
      ),
      header = TRUE,
      sep = "\t"
    )
    samples <- fread(
      paste0(
        "../analyses/results/datasets/historical-",
        f,
        "/poplists/historical-",
        f,
        "_all.indiv.list"
      ),
      header = TRUE,
      sep = "\t",
      select = c("sample", "population", "time", "depth")
    )
    hz <- merge(hz, samples, by = "sample")
    hz <- merge(hz, qc, by = "sample")
    if (s %in% species) {
      hz$year <- str_sub(hz$population, 1, 4)
      hz$region <- "one_region"
    } else {
      hz$year <- str_sub(hz$population, -4, -1)
      hz$region <- str_sub(hz$population, 1, -5)
      # Lump SW Skåne with W Skåne for Pl. argus
      hz[hz$region == "SWSkane", ]$region <- "WSkane"
    }
    if (nrow(hz) > 0) {
      regions <- unique(hz$region)

      relhet <- c()

      for (r in regions) {
        if (r == "SESkane") {
          historical <- hz[hz$region == "ESkane" & hz$year < 1995, ]$sample
        } else {
          historical <- hz[hz$region == r & hz$year < 1995, ]$sample
        }
        modern <- hz[hz$region == r & hz$year > 1995, ]$sample
        if (length(historical) > 0 & length(modern) > 0) {
          combos <- expand.grid(historical, modern)
          colnames(combos) <- c("historical", "modern")
          combos <- merge(
            combos,
            hz[, c("sample", "hetperbp", "year")],
            by.x = "historical", by.y = "sample"
          )
          colnames(combos) <- c("historical", "modern", "histhz", "histyear")
          combos <- merge(
            combos,
            hz[, c("sample", "hetperbp", "year")],
            by.x = "modern", by.y = "sample"
          )
          colnames(combos) <- c(
            "historical", "modern", "histhz", "histyear", "modhz",
            "modyear"
          )
          # below is the main estimate of heterozygosity retention per year
          combos$histyear <- as.numeric(combos$histyear)
          combos$modyear <- as.numeric(combos$modyear)
          combos$het_retained <- (combos$modhz / combos$histhz)
          combos$het_retained_peryear <- exp(
            log(combos$het_retained) / (combos$modyear - combos$histyear)
          )
          combos$rel_delta_het <- combos$het_retained - 1
          combos$rel_delta_het_peryear <- combos$het_retained_peryear - 1
          combos$pred_het_ret_100 <- (combos$het_retained_peryear^100) * 100
          if (s %in% c(
            "aagestis", "cminimus", "palcon", "pamandus", "pargyrognomon"
          )) {
            r <- "WSkane"
          } else if (s == "aoptilete") {
            r <- "NSmaland"
          } else if (s == "cargiolus") {
            r <- "Oland"
          }
          combos$region <- r
          combos$species <- s
          relhet_all <- rbind(relhet_all, combos)
          relhet_all$historical <- as.character(relhet_all$historical)
          relhet_all$modern <- as.character(relhet_all$modern)
        } else {
          df <- data.frame(
            "hist", "mod", 0, 0, 0, 0, 0, 0, 0, 0, 0, "other_region", s
          )
          colnames(df) <- colnames(relhet_all)
          relhet_all <- rbind(relhet_all, df)
        }
      }
    } else {
      df <- data.frame(
        "hist", "mod", 0, 0, 0, 0, 0, 0, 0, 0, 0, "other_region", s
      )
      colnames(df) <- colnames(relhet_all)
      relhet_all <- rbind(relhet_all, df)
    }
  }

relhet_meds <- relhet_all %>%
  group_by(species, region) %>%
  summarise(
    med = median(pred_het_ret_100),
    min = min(pred_het_ret_100),
    max = max(pred_het_ret_100)
  )
